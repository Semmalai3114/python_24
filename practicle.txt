in master:

kubectl get nodes
kubectl run nginx-pod --image=nginx(create pods)
kubectl run nginx1 --image=nginx -o yaml --dry-run=client(note create but show the status of pod with yaml)

kubectl run nginx1 --image=nginx -o yaml > pads.yaml(save pods cods to yaml file)

entry and cmd defferents:
create Dockerfile:
FROM ubuntu
ENTRYPOINT ["/bin/echo"]
CMD ["hello","world"]
 docker build -t imagename .
docker run imagename -->hello world
docker run imagename how are you! -->how are you! (override the cmd)

in kubernets:
kubectl run echo1 --image=busybox:latest --restart=Never --command  -- echo "hello world"
manifest file:

apiVersion: v1
kind: Pod
metadata:
  name: nginx-pod
spec:
  containers:
    - name: nginx-container
      image: nginx
      command: ["/bin/echo"]
      args: ["hello", "world"]


kubectl logs nginx-pod

replicaSet;
apiVersion: apps/v1
kind: ReplicaSet
metadata:
  name: frontend
spec:
  # modify replicas according to your case
  replicas: 3
  selector:
    matchLabels:
      tier: frontend
  template:
    metadata:
      labels:
        tier: frontend
    spec:
      containers:
      - name: php-redis
        image: us-docker.pkg.dev/google-samples/containers/gke/gb-frontend:v5
