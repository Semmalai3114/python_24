FROM ubuntu
ENTRYPOINT ["/bin/echo"]
CMD ["hello","world"]

docker run imagename how are you! -->how are you! (override the cmd)

kubectl run echo1 --image=busybox:latest --restart=Never --command  -- echo "hello world"

replicaSet;
apiVersion: apps/v1
kind: ReplicaSet
metadata:
  name: frontend
spec:
  # modify replicas according to your case
  replicas: 3
  selector:
    matchLabels:
      tier: frontend
  template:
    metadata:
      labels:
        tier: frontend
    spec:
      containers:
      - name: php-redis
        image: us-docker.pkg.dev/google-samples/containers/gke/gb-frontend:v5

increassing setsize(vm count);
kubectl scale --replicas=5 rs/rs-name

decrease scale:
kubectl scale --replicas=1 rs/rs-name

deployement:
kubectl create deployment nginx-deploy  --image=nginx --replicas=3 
kubectl apply -f deployment.yaml
kubectl get deployment

kubectl get pods
reduce or increace scale
kubectl scale deployment nginx-deploy --replicas=0

history:
kubectl rollout history deployment deployement-name

previous version:
kubectl undo rollout deployment deployment-name 


node:
kubectl create deployment nginx-deploy  --image=nginx --replicas=3 --dry-run=client > nginx-deploy.yaml
kubectl apply nginx-deploy.yaml
kubectl get pods -o wide(pods will created in separate nodes)
 

remove nodes

kubectl delete node node-name (pod in that node will delete and recreate in other live node with free space )

nodeSelectore:(under in template -->spec -->nodeSelector:)

kubectl label node node-name size=small


template:
    metadata:
      creationTimestamp: null
      labels:
        app: nginx-deploy
    spec:
      nodeSelector:
	size: small(create pods in particular nodes using label name)
      containers:
      - image: nginx
        imagePullPolicy: Always
        name: nginx
        resources: {}
        terminationMessagePath: /dev/termination-log
        terminationMessagePolicy: File
      dnsPolicy: ClusterFirst
      restartPolicy: Always
      schedulerName: default-scheduler
      securityContext: {}
      terminationGracePeriodSeconds: 30


kubectl apply nginx-deploy.yaml

kubectl get pods -o wide
